On $\bbZ^{d},$ let $W_t$ be the matrix with entries $(W_t)_{xy} = \delta_{xy} W_{t,x}$, where each $W_{t,x}$ is an independent standard Brownian motion.

Let $H_t = \Delta + \lambda W_t$. For $E_0 \in  \bbR$ fixed, let
\begin{align*}
z_t = E_0  + (1-t) \lambda^{2} \theta(E_0),
\end{align*}
and let $R_t = (H_t - z_t)^{-1}$ be the resolvent of $H_t$ at this spectral parameter.
Here, for $z\in \bbH$, $\theta(z)$ is defined to be the unique solution to be the unique solution of the self-consistent equation
\begin{align*}
\left(\Delta  - (z  + \lambda^{2} \theta(z)) \right)^{-1}_{00} = \theta(z),
\end{align*}
whereas for $E \in R$ we let $\theta(E) = \lim_{\eta \to 0}\theta(E+ i\eta)$.

For $x\in  \bbZ^{d}$, let $S_{x}$ be the matrix $(S_{x})_{ab} = \delta_{ax} \delta_{bx}$. 
\AB{This is a choice we make for the sake of concreteness and may later find that we instead want to consider observables that average over the kinetic scale.}
In the parlance of [ER] we focus on \emph{saturated chains} so for $p<j\leq k$ and $\bs x = (x_1,\ldots,x_k) \in  (\bbZ^{d})^{k}$ we merely define
\begin{align*}
R_{[j,p],t}(\bs x)  = R_t^{\pm} S^{x_p} R_t^{\pm} S^{x_{p+1}} \cdots S^{x_{j-1}} R_t^{\pm}
\end{align*}
where $R_t^{+} = R_t$ and $R_t^{-} = R_t^{*}$, and the signs always alternate starting with $+$ when $p$ is odd and $-$ when $p$ is even. 
We think of this as a matrix-valued function of $\bs x$.

We use the convention that for $k \in \bbN$, $R_{t,k} =  R_{t,(+,-,\ldots,+,-)}$.

Let  $\calD[\left( \cdot \right)]:= \sum_{x} S_x \left( \cdot \right)  S_x$ be the \emph{diagonal superoperator} so that for a matrix $A$, $(\calD[A])_{xy} = \delta_{xy} A_{xy}$.

\begin{proposition}
The matrix $R_{t}$ solves the SDE
\begin{align*}
dR_t = \lambda^{2} R_t \left( \calD[R_t] - \theta(E_0)  \right)  R_t \: dt  -  \lambda R_t\: dW_t\: R_t.
\end{align*}
\end{proposition}
\begin{proof}
Using that
\begin{align*}
\frac{dR_t}{dt} &= -\lambda^{2} \theta(E_0) R^{2}_t\\
\frac{dR_t}{dW_x} &= - \lambda R_t S_x R_t\\
\frac{d^{2}R_t}{dW_xdW_y} &=  \lambda^{2}\left(  R_t S_x R_t S_y R + R_t S_x R_t S_y R_t \right)
\end{align*}
the result follows immediately from It\^{o}'s formula.
\end{proof}

Ignoring the second term, we observe that a solution to the ODE
\begin{align*}
dM_t = \lambda^{2} M_t \left( \calD[M_t] - \theta(E_0) \right) M_t \: dt
\end{align*}
is the constant in time matrix $M_{t} = \left(\Delta - (E_0 + \lambda^{2} \theta(E_0)\right)^{-1}$

\section{Two loops}
Consider the second order \emph{isotropic} chain
\begin{align*}
R_{[1,2],t} = R_t S^{x_1} R_t^{*}.
\end{align*}
Isotropic here means that we still think of this chain as a matrix, rather than the averaged $2$-chain
\begin{align*}
\Tr R_t S^{x_1} R_t^{*} S^{x_2}.
\end{align*}
We expect to have to consider both eventually.

By It\^{o} calculus as above, we have the following
\begin{proposition}
The $2$-chain $R_{[1,2],t}$ solves the SDE
\begin{align*}
d R_{[1,2],t} = \lambda^{2} \Theta_t[R_{[1,2],t}] \: dt
\end{align*}
\end{proposition}
